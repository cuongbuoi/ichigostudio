# Ichigo Studio BE + Admin (Phase 0 & 1) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Dựng 2 repo — `ichigo-studio-api` (Cloudflare Workers) và `ichigo-studio-admin` (Vue 3) — cho phép chủ shop đăng nhập, quản lý (CRUD) sản phẩm và upload ảnh, kèm endpoint public đọc-only cho storefront.

**Architecture:** API là Hono trên Cloudflare Workers, dữ liệu ở D1 (SQLite qua Drizzle), ảnh ở R2, auth bằng JWT HS256 với một tài khoản admin seed sẵn (mật khẩu băm PBKDF2 Web Crypto). Admin là SPA Vue 3 + Vite + shadcn-vue gọi API qua Bearer token, dùng Pinia (auth) + TanStack Query (data).

**Tech Stack:** Hono, `@hono/zod-validator`, Drizzle ORM, `wrangler`, D1, R2, `hono/jwt`, Web Crypto PBKDF2, vitest — cho API. Vue 3, Vite, TypeScript, Tailwind v3, shadcn-vue, vue-router, Pinia, `@tanstack/vue-query`, vee-validate + zod, `ofetch`, vitest + @vue/test-utils — cho Admin.

```typescript
const ITERATIONS = 100_000
const enc = new TextEncoder()

function toB64(buf: ArrayBuffer): string {
  return btoa(String.fromCharCode(...new Uint8Array(buf)))
}
function fromB64(s: string): Uint8Array {
  return Uint8Array.from(atob(s), (c) => c.charCodeAt(0))
}

async function derive(plain: string, salt: Uint8Array): Promise<ArrayBuffer> {
  const key = await crypto.subtle.importKey('raw', enc.encode(plain), 'PBKDF2', false, ['deriveBits'])
  return crypto.subtle.deriveBits(
    { name: 'PBKDF2', salt, iterations: ITERATIONS, hash: 'SHA-256' },
    key,
    256,
  )
}

export async function hashPassword(plain: string): Promise<string> {
  const salt = crypto.getRandomValues(new Uint8Array(16))
  const bits = await derive(plain, salt)
  return `pbkdf2$${ITERATIONS}$${toB64(salt.buffer)}$${toB64(bits)}`
}

export async function verifyPassword(plain: string, stored: string): Promise<boolean> {
  const [scheme, iterStr, saltB64, hashB64] = stored.split('$')
  if (scheme !== 'pbkdf2') return false
  const salt = fromB64(saltB64)
  const key = await crypto.subtle.importKey('raw', enc.encode(plain), 'PBKDF2', false, ['deriveBits'])
  const bits = await crypto.subtle.deriveBits(
    { name: 'PBKDF2', salt, iterations: Number(iterStr), hash: 'SHA-256' },
    key,
    256,
  )
  const a = new Uint8Array(bits)
  const b = fromB64(hashB64)
  if (a.length !== b.length) return false
  let diff = 0
  for (let i = 0; i < a.length; i++) diff |= a[i] ^ b[i]
  return diff === 0
}
```

```typescript
import { sign, verify } from 'hono/jwt'

const EXPIRES_SECONDS = 60 * 60 * 24 * 7 // 7 ngày

export async function signToken(userId: string, secret: string): Promise<string> {
  const now = Math.floor(Date.now() / 1000)
  return sign({ sub: userId, iat: now, exp: now + EXPIRES_SECONDS }, secret)
}

export async function verifyToken(token: string, secret: string): Promise<{ sub: string } | null> {
  try {
    const payload = await verify(token, secret)
    return { sub: String(payload.sub) }
  } catch {
    return null
  }
}
```

```typescript
import { createMiddleware } from 'hono/factory'
import type { Bindings } from '../index'
import { verifyToken } from '../auth/jwt'

export const requireAuth = createMiddleware<{
  Bindings: Bindings
  Variables: { userId: string }
}>(async (c, next) => {
  const header = c.req.header('authorization') ?? ''
  const token = header.startsWith('Bearer ') ? header.slice(7) : ''
  const payload = token ? await verifyToken(token, c.env.JWT_SECRET) : null
  if (!payload) return c.json({ error: 'unauthorized' }, 401)
  c.set('userId', payload.sub)
  await next()
})
```

```typescript
import { Hono } from 'hono'
import { zValidator } from '@hono/zod-validator'
import { z } from 'zod'
import { eq } from 'drizzle-orm'
import type { Bindings } from '../index'
import { getDb } from '../db/client'
import { users } from '../db/schema'
import { verifyPassword } from '../auth/password'
import { signToken } from '../auth/jwt'
import { requireAuth } from '../middleware/require-auth'

const loginSchema = z.object({ email: z.string().email(), password: z.string().min(1) })

export const authRoutes = new Hono<{ Bindings: Bindings; Variables: { userId: string } }>()

authRoutes.post('/login', zValidator('json', loginSchema), async (c) => {
  const { email, password } = c.req.valid('json')
  const db = getDb(c.env.DB)
  const rows = await db.select().from(users).where(eq(users.email, email)).limit(1)
  const user = rows[0]
  if (!user || !(await verifyPassword(password, user.passwordHash))) {
    return c.json({ error: 'invalid_credentials' }, 401)
  }
  const token = await signToken(user.id, c.env.JWT_SECRET)
  return c.json({ token })
})

authRoutes.get('/me', requireAuth, async (c) => {
  const db = getDb(c.env.DB)
  const rows = await db.select().from(users).where(eq(users.id, c.get('userId'))).limit(1)
  const user = rows[0]
  if (!user) return c.json({ error: 'not_found' }, 404)
  return c.json({ id: user.id, email: user.email })
})
```

```typescript
import { Hono } from 'hono'
import { zValidator } from '@hono/zod-validator'
import { and, eq } from 'drizzle-orm'
import { z } from 'zod'
import type { Bindings } from '../index'
import { getDb } from '../db/client'
import { products, productImages } from '../db/schema'
import { requireAuth } from '../middleware/require-auth'
import { productInputSchema, productPatchSchema } from '../schemas/product'

export const productRoutes = new Hono<{ Bindings: Bindings; Variables: { userId: string } }>()

productRoutes.use('*', requireAuth)

const listQuery = z.object({ series: z.string().optional(), status: z.enum(['draft', 'published']).optional() })

productRoutes.get('/', zValidator('query', listQuery), async (c) => {
  const { series, status } = c.req.valid('query')
  const db = getDb(c.env.DB)
  const conds = []
  if (series) conds.push(eq(products.series, series))
  if (status) conds.push(eq(products.status, status))
  const items = await db.select().from(products).where(conds.length ? and(...conds) : undefined)
  return c.json({ items })
})

async function loadImages(db: ReturnType<typeof getDb>, productId: string) {
  return db.select().from(productImages).where(eq(productImages.productId, productId))
}

productRoutes.get('/:id', async (c) => {
  const db = getDb(c.env.DB)
  const rows = await db.select().from(products).where(eq(products.id, c.req.param('id'))).limit(1)
  const product = rows[0]
  if (!product) return c.json({ error: 'not_found' }, 404)
  return c.json({ ...product, images: await loadImages(db, product.id) })
})

async function replaceImages(db: ReturnType<typeof getDb>, productId: string, urls: string[]) {
  await db.delete(productImages).where(eq(productImages.productId, productId))
  if (urls.length) {
    await db.insert(productImages).values(
      urls.map((url, i) => ({ id: crypto.randomUUID(), productId, url, sort: i })),
    )
  }
}

productRoutes.post('/', zValidator('json', productInputSchema), async (c) => {
  const input = c.req.valid('json')
  const db = getDb(c.env.DB)
  const id = crypto.randomUUID()
  const now = Date.now()
  const { images, ...fields } = input
  await db.insert(products).values({ id, ...fields, createdAt: now, updatedAt: now })
  await replaceImages(db, id, images)
  return c.json({ id }, 201)
})

productRoutes.patch('/:id', zValidator('json', productPatchSchema), async (c) => {
  const id = c.req.param('id')
  const input = c.req.valid('json')
  const db = getDb(c.env.DB)
  const rows = await db.select().from(products).where(eq(products.id, id)).limit(1)
  if (!rows[0]) return c.json({ error: 'not_found' }, 404)
  const { images, ...fields } = input
  await db.update(products).set({ ...fields, updatedAt: Date.now() }).where(eq(products.id, id))
  if (images) await replaceImages(db, id, images)
  const updated = await db.select().from(products).where(eq(products.id, id)).limit(1)
  return c.json({ ...updated[0], images: await loadImages(db, id) })
})

productRoutes.delete('/:id', async (c) => {
  const db = getDb(c.env.DB)
  const id = c.req.param('id')
  await db.delete(products).where(eq(products.id, id))
  return c.json({ ok: true })
})
```

```typescript
import { Hono } from 'hono'
import type { Bindings } from '../index'
import { requireAuth } from '../middleware/require-auth'

export const uploadRoutes = new Hono<{ Bindings: Bindings; Variables: { userId: string } }>()

uploadRoutes.post('/', requireAuth, async (c) => {
  const form = await c.req.formData()
  const file = form.get('file')
  if (!(file instanceof File)) return c.json({ error: 'file_required' }, 400)
  const safeName = file.name.replace(/[^a-zA-Z0-9._-]/g, '_')
  const key = `products/${crypto.randomUUID()}-${safeName}`
  await c.env.BUCKET.put(key, await file.arrayBuffer(), {
    httpMetadata: { contentType: file.type || 'application/octet-stream' },
  })
  const origin = new URL(c.req.url).origin
  return c.json({ key, url: `${origin}/${key}` }, 201)
})
```

```typescript
import { cors } from 'hono/cors'
import { uploadRoutes } from './routes/uploads'

// sau khi tạo app:
app.use('*', async (c, next) => {
  const allowed = c.env.ALLOWED_ORIGINS.split(',').map((s) => s.trim())
  return cors({
    origin: (origin) => (allowed.includes(origin) ? origin : allowed[0]),
    allowMethods: ['GET', 'POST', 'PATCH', 'DELETE', 'OPTIONS'],
    allowHeaders: ['Authorization', 'Content-Type'],
  })(c, next)
})

app.onError((err, c) => {
  console.error(err)
  return c.json({ error: 'internal_error' }, 500)
})

// mount uploads + serve R2 objects
app.route('/uploads', uploadRoutes)
app.get('/products/:key{.+}', async (c) => {
  // chỉ serve object trong prefix products/ đã upload
  const key = `products/${c.req.param('key')}`
  const obj = await c.env.BUCKET.get(key)
  if (!obj) return c.notFound()
  c.header('Content-Type', obj.httpMetadata?.contentType ?? 'application/octet-stream')
  c.header('Cache-Control', 'public, max-age=31536000, immutable')
  return c.body(obj.body)
})
```

Note thứ tự mount: CORS + onError đặt NGAY sau khi tạo `app`, TRƯỚC các `app.route(...)`. import { defineStore } from 'pinia'
```typescript
import { ref, computed } from 'vue'
import { api, TOKEN_KEY } from '@/lib/api'

export const useAuth = defineStore('auth', () => {
  const token = ref<string | null>(localStorage.getItem(TOKEN_KEY))
  const isAuthed = computed(() => !!token.value)

  async function login(email: string, password: string) {
    const res = await api<{ token: string }>('/auth/login', {
      method: 'POST',
      body: { email, password },
    })
    token.value = res.token
    localStorage.setItem(TOKEN_KEY, res.token)
  }

  function logout() {
    token.value = null
    localStorage.removeItem(TOKEN_KEY)
  }

  return { token, isAuthed, login, logout }
})
```

```typescript
import { api } from '@/lib/api'

export interface Product {
  id: string
  slug: string
  title: string
  series: string
  material: string
  priceRef?: string | null
  cover?: string | null
  featured: boolean
  order: number
  status: 'draft' | 'published'
  bodyMd?: string | null
  images?: string[]
}

export interface ProductInput {
  slug: string
  title: string
  series: string
  material: string
  priceRef?: string
  cover?: string
  featured: boolean
  order: number
  status: 'draft' | 'published'
  bodyMd?: string
  images: string[]
}

export function fetchProducts(params?: { series?: string; status?: 'draft' | 'published' }) {
  return api<{ items: Product[] }>('/products', { query: params ?? {} })
}

export function fetchProduct(id: string) {
  return api<Product & { images: { url: string }[] }>(`/products/${id}`)
}

export function createProduct(input: ProductInput) {
  return api<{ id: string }>('/products', { method: 'POST', body: input })
}

export function updateProduct(id: string, input: Partial<ProductInput>) {
  return api<Product>(`/products/${id}`, { method: 'PATCH', body: input })
}

export function deleteProduct(id: string) {
  return api<{ ok: true }>(`/products/${id}`, { method: 'DELETE' })
}

export async function uploadImage(file: File): Promise<string> {
  const fd = new FormData()
  fd.append('file', file)
  const res = await api<{ url: string }>('/uploads', { method: 'POST', body: fd })
  return res.url
}
```

```vue
<script setup lang="ts">
import { ref, onMounted, reactive } from 'vue'
import { useRoute, useRouter } from 'vue-router'
import { fetchProduct, createProduct, updateProduct, uploadImage } from '@/lib/products'
import { productFormSchema } from '@/lib/product-schema'
import { Button } from '@/components/ui/button'
import { Input } from '@/components/ui/input'
import { Label } from '@/components/ui/label'

const route = useRoute()
const router = useRouter()
const id = route.params.id as string | undefined
const isEdit = !!id

const form = reactive({
  slug: '', title: '', series: '', material: '', priceRef: '',
  cover: '', featured: false, order: 0,
  status: 'draft' as 'draft' | 'published', bodyMd: '',
  images: [] as string[],
})
const errors = ref<Record<string, string>>({})
const saving = ref(false)

onMounted(async () => {
  if (!isEdit) return
  const p = await fetchProduct(id!)
  Object.assign(form, {
    slug: p.slug, title: p.title, series: p.series, material: p.material,
    priceRef: p.priceRef ?? '', cover: p.cover ?? '', featured: p.featured,
    order: p.order, status: p.status, bodyMd: p.bodyMd ?? '',
    images: (p.images ?? []).map((i) => i.url),
  })
})

async function onUpload(e: Event) {
  const files = (e.target as HTMLInputElement).files
  if (!files) return
  for (const f of Array.from(files)) {
    const url = await uploadImage(f)
    form.images.push(url)
    if (!form.cover) form.cover = url
  }
}

function removeImage(i: number) { form.images.splice(i, 1) }

async function onSubmit() {
  errors.value = {}
  const parsed = productFormSchema.safeParse({ ...form })
  if (!parsed.success) {
    for (const issue of parsed.error.issues) errors.value[String(issue.path[0])] = issue.message
    return
  }
  saving.value = true
  try {
    if (isEdit) await updateProduct(id!, parsed.data)
    else await createProduct(parsed.data)
    router.push({ name: 'products' })
  } finally {
    saving.value = false
  }
}
</script>

<template>
  <div class="min-h-screen bg-night text-paper p-6">
    <h1 class="text-2xl font-semibold mb-6">{{ isEdit ? 'Sửa sản phẩm' : 'Tạo sản phẩm' }}</h1>
    <form class="max-w-2xl space-y-4" @submit.prevent="onSubmit">
      <div class="space-y-2"><Label>Tên</Label><Input v-model="form.title" /><p v-if="errors.title" class="text-sm text-red-400">{{ errors.title }}</p></div>
      <div class="space-y-2"><Label>Slug</Label><Input v-model="form.slug" placeholder="set-day-noi" /><p v-if="errors.slug" class="text-sm text-red-400">{{ errors.slug }}</p></div>
      <div class="grid grid-cols-2 gap-4">
        <div class="space-y-2"><Label>Series</Label><Input v-model="form.series" /><p v-if="errors.series" class="text-sm text-red-400">{{ errors.series }}</p></div>
        <div class="space-y-2"><Label>Chất liệu</Label><Input v-model="form.material" /><p v-if="errors.material" class="text-sm text-red-400">{{ errors.material }}</p></div>
      </div>
      <div class="grid grid-cols-2 gap-4">

        <div class="space-y-2"><Label>Thứ tự</Label><Input v-model.number="form.order" type="number" /></div>
      </div>
      <div class="flex items-center gap-6">
        <label class="flex items-center gap-2"><input type="checkbox" v-model="form.featured" /> Nổi bật</label>
        <label class="flex items-center gap-2">Trạng thái
          <select v-model="form.status" class="bg-panel border rounded px-2 py-1">
```
